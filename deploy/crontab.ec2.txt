# Infinitee Xclusives — EC2 (Ubuntu) crontab
# ─────────────────────────────────────────────────────────────────────────────
# WHY THIS FILE EXISTS
#   deploy/crontab.txt is a macOS-LOCAL crontab: it points at
#   /Users/farhankauser/Desktop/project/infinitee_app and an Anaconda python,
#   neither of which exists on the server. If it was ever installed on EC2,
#   every job silently failed with "No such file or directory" — including
#   `ingest_settlement_reports`, which is what keeps the FBA actual-fee data
#   (SkuFeeActual) current. This is the server-side equivalent.
#
# INSTALL (on EC2):
#   crontab ~/Daily-Sales-Dashboard/deploy/crontab.ec2.txt
# VERIFY:
#   crontab -l
#   ls -l ~/Daily-Sales-Dashboard/logs/settlement.log
#
# Times are in the server's timezone (UTC unless changed). Adjust if the
# business wants these anchored to PST.
# ─────────────────────────────────────────────────────────────────────────────

SHELL=/bin/bash
PATH=/usr/local/sbin:/usr/local/bin:/usr/sbin:/usr/bin:/sbin:/bin
APP=/home/ubuntu/Daily-Sales-Dashboard
PYTHON=/home/ubuntu/Daily-Sales-Dashboard/venv/bin/python
MANAGE=/home/ubuntu/Daily-Sales-Dashboard/manage.py
LOG=/home/ubuntu/Daily-Sales-Dashboard/logs
MAILTO=""

# ── Core metrics ─────────────────────────────────────────────────────────────
5 * * * *    cd $APP && $PYTHON $MANAGE snapshot_hourly_metrics >> $LOG/hourly.log 2>&1
0 10 * * *   cd $APP && $PYTHON $MANAGE sync_daily_metrics      >> $LOG/daily.log 2>&1
45 0 * * *   cd $APP && $PYTHON $MANAGE finalize_yesterday      >> $LOG/finalize.log 2>&1

# ── Settlement ingest → SkuFeeActual (FBA fee drift "actual") ────────────────
# Amazon generates settlements on ~14-day pay cycles, so most runs find nothing
# new — that is expected and cheap. Twice daily so a newly-DONE report is picked
# up the same day rather than up to 24h later.
30 4 * * *   cd $APP && $PYTHON $MANAGE ingest_settlement_reports >> $LOG/settlement.log 2>&1
30 16 * * *  cd $APP && $PYTHON $MANAGE ingest_settlement_reports >> $LOG/settlement.log 2>&1
# Weekly wider sweep — catches any report that appeared late or was missed
# while the server was down. --days 90 re-lists older reports; ingestion is
# idempotent (dedupes on SettlementReport.report_id) so this is safe.
0 5 * * 0    cd $APP && $PYTHON $MANAGE ingest_settlement_reports --days 90 >> $LOG/settlement.log 2>&1

# ── PPC / Ads ────────────────────────────────────────────────────────────────
0 6 * * *    cd $APP && $PYTHON $MANAGE backfill_ppc --marketplace usa --days 7 >> $LOG/ppc_daily.log 2>&1
*/30 * * * * cd $APP && $PYTHON $MANAGE sync_today_ppc --marketplace usa       >> $LOG/ppc_today.log 2>&1
15 * * * *   cd $APP && $PYTHON $MANAGE compute_sku_ppc --days 1              >> $LOG/sku_ppc.log 2>&1
0 3 * * *    cd $APP && $PYTHON $MANAGE compute_sku_ppc --days 4              >> $LOG/sku_ppc.log 2>&1
0 4 * * 0    cd $APP && $PYTHON $MANAGE compute_sku_ppc --backfill            >> $LOG/sku_ppc.log 2>&1
* * * * *    cd $APP && $PYTHON $MANAGE ingest_ams_s3 --max-objects 100       >> $LOG/ams_ingest.log 2>&1
30 5 * * *   cd $APP && $PYTHON $MANAGE seed_ams_subscriptions                >> $LOG/ams_subs.log 2>&1
30 3 * * *   cd $APP && $PYTHON $MANAGE ingest_ads_detail_reports             >> $LOG/ads_detail.log 2>&1
# NOTE: the macOS file used `date -v-1d` (BSD). GNU date on Ubuntu needs -d.
*/15 8-23 * * * cd $APP && $PYTHON $MANAGE ingest_ads_detail_reports --date $(date -d 'yesterday' +\%Y-\%m-\%d) >> $LOG/ads_detail.log 2>&1
30 6 * * *   cd $APP && $PYTHON $MANAGE ingest_ads_detail_reports --rewind 6 --report-kinds sp_search_term,sb_search_term,sp_advertised_product,sb_advertised_product,sd_advertised_product >> $LOG/ads_detail.log 2>&1

# ── Derived / intelligence ───────────────────────────────────────────────────
0 5 * * *    cd $APP && $PYTHON $MANAGE compute_campaign_profit      >> $LOG/campaign_profit.log 2>&1
30 5 * * *   cd $APP && $PYTHON $MANAGE generate_profit_alerts       >> $LOG/profit_alerts.log 2>&1
0 6 * * *    cd $APP && $PYTHON $MANAGE generate_ai_recommendations  >> $LOG/ai_recommendations.log 2>&1

# ── Housekeeping ─────────────────────────────────────────────────────────────
30 3 * * 0   cd $APP && $PYTHON $MANAGE prune_hourly_snapshots --keep-days 30 >> $LOG/prune.log 2>&1
45 3 * * 0   cd $APP && $PYTHON $MANAGE prune_daily_metrics   --keep-days 90 >> $LOG/prune.log 2>&1
